CREATE VIEW v_alertas_observadas AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, tolerancia_tasa, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now'), '-' || (horizonte_horas + 1) || ' hours') AS desde_txt, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now')) AS hasta_txt FROM alerta_parametros WHERE id = 1 ), c AS ( SELECT v.metrica_id, v.prediccion_id, v.ventana_inicio, v.ventana_fin, v.duracion_ventana_minutos, v.ambiente, v.cliente_id, v.servicio_id, v.producto_id, v.canal_id, v.moneda_id, v.cantidad_total, v.cantidad_exitos, v.cantidad_rechazos, v.cantidad_errores, v.tasa_aprobacion_calc, v.tasa_rechazo_calc, v.tasa_error_calc, v.tasa_aprobacion_informada, v.tasa_rechazo_informada, v.tasa_error_informada, v.importe_total, v.tiempo_respuesta_promedio, v.tiempo_respuesta_maximo_obs, v.ultimo_evento_exitoso, v.fecha_recepcion, v.origen_dato, v.cantidad_total_esperada, CASE WHEN COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) = 1 OR v.cantidad_total_esperada IS NULL THEN v.cantidad_total_minima ELSE v.cantidad_total_esperada + COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) * (v.cantidad_total_minima - v.cantidad_total_esperada) END AS cantidad_total_minima, CASE WHEN COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) = 1 OR v.cantidad_total_esperada IS NULL THEN v.cantidad_total_maxima ELSE v.cantidad_total_esperada + COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) * (v.cantidad_total_maxima - v.cantidad_total_esperada) END AS cantidad_total_maxima, v.tasa_aprobacion_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR v.tasa_aprobacion_esperada IS NULL THEN v.tasa_aprobacion_minima ELSE v.tasa_aprobacion_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (v.tasa_aprobacion_minima - v.tasa_aprobacion_esperada) END AS tasa_aprobacion_minima, v.tasa_rechazo_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR v.tasa_rechazo_esperada IS NULL THEN v.tasa_rechazo_maxima ELSE v.tasa_rechazo_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (v.tasa_rechazo_maxima - v.tasa_rechazo_esperada) END AS tasa_rechazo_maxima, v.tasa_error_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR v.tasa_error_esperada IS NULL THEN v.tasa_error_maxima ELSE v.tasa_error_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (v.tasa_error_maxima - v.tasa_error_esperada) END AS tasa_error_maxima, v.importe_total_esperado, CASE WHEN COALESCE(us.banda_importe, uc.banda_importe, ug.banda_importe, 1.0) = 1 OR v.importe_total_esperado IS NULL THEN v.importe_total_minimo ELSE v.importe_total_esperado + COALESCE(us.banda_importe, uc.banda_importe, ug.banda_importe, 1.0) * (v.importe_total_minimo - v.importe_total_esperado) END AS importe_total_minimo, CASE WHEN COALESCE(us.banda_importe, uc.banda_importe, ug.banda_importe, 1.0) = 1 OR v.importe_total_esperado IS NULL THEN v.importe_total_maximo ELSE v.importe_total_esperado + COALESCE(us.banda_importe, uc.banda_importe, ug.banda_importe, 1.0) * (v.importe_total_maximo - v.importe_total_esperado) END AS importe_total_maximo, v.tiempo_respuesta_esperado, CASE WHEN COALESCE(us.banda_latencia, uc.banda_latencia, ug.banda_latencia, 1.0) = 1 OR v.tiempo_respuesta_esperado IS NULL THEN v.tiempo_respuesta_maximo_esp ELSE v.tiempo_respuesta_esperado + COALESCE(us.banda_latencia, uc.banda_latencia, ug.banda_latencia, 1.0) * (v.tiempo_respuesta_maximo_esp - v.tiempo_respuesta_esperado) END AS tiempo_respuesta_maximo_esp, v.modelo_id, v.modelo_version, COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) AS volumen_minimo, COALESCE(us.gracia_minutos, uc.gracia_minutos, ug.gracia_minutos, prm.gracia_minutos) AS gracia_minutos, prm.tolerancia_tasa, (julianday(v.fecha_recepcion) - julianday(v.ventana_fin)) * 1440 AS demora_minutos, COALESCE(us.corte_pct_alta, uc.corte_pct_alta, ug.corte_pct_alta, 0.25) AS corte_pct_alta, COALESCE(us.corte_pct_critica, uc.corte_pct_critica, ug.corte_pct_critica, 0.50) AS corte_pct_critica, COALESCE(us.corte_tasa_alta, uc.corte_tasa_alta, ug.corte_tasa_alta, 0.05) AS corte_tasa_alta, COALESCE(us.corte_tasa_critica, uc.corte_tasa_critica, ug.corte_tasa_critica, 0.15) AS corte_tasa_critica FROM prm CROSS JOIN v_comparacion v LEFT JOIN umbrales_config ug ON ug.cliente_id = '*' AND ug.servicio_id = '*' LEFT JOIN umbrales_config uc ON uc.cliente_id = v.cliente_id AND uc.servicio_id = '*' LEFT JOIN umbrales_config us ON us.cliente_id = v.cliente_id AND us.servicio_id = v.servicio_id LEFT JOIN feriados fe ON fe.fecha = date(v.ventana_inicio, '-3 hours') WHERE NOT (fe.fecha IS NOT NULL AND COALESCE(us.silenciar_feriados, uc.silenciar_feriados, ug.silenciar_feriados, 0) = 1) AND v.ventana_inicio >= prm.desde_txt AND v.ventana_inicio < prm.hasta_txt AND julianday(v.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND julianday(v.ventana_fin) <= prm.ahora_jd ), e AS ( SELECT c.*, r.regla, CASE r.regla WHEN 'VOLUMEN_BAJO' THEN c.cantidad_total_minima IS NOT NULL AND c.cantidad_total < c.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN c.cantidad_total_maxima IS NOT NULL AND c.cantidad_total > c.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_aprobacion_minima IS NOT NULL AND c.tasa_aprobacion_calc < c.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_rechazo_maxima IS NOT NULL AND c.tasa_rechazo_calc > c.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_error_maxima IS NOT NULL AND c.tasa_error_calc > c.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN (c.importe_total_minimo IS NOT NULL AND c.importe_total < c.importe_total_minimo) OR (c.importe_total_maximo IS NOT NULL AND c.importe_total > c.importe_total_maximo) WHEN 'LATENCIA_ALTA' THEN c.tiempo_respuesta_maximo_esp IS NOT NULL AND c.tiempo_respuesta_promedio > c.tiempo_respuesta_maximo_esp WHEN 'SIN_EXITOS' THEN c.cantidad_total >= c.volumen_minimo AND c.cantidad_exitos = 0 WHEN 'DATO_TARDIO' THEN c.origen_dato = 'API' AND c.demora_minutos > c.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN c.cantidad_total > 0 AND ( ABS(COALESCE(c.tasa_aprobacion_informada, c.tasa_aprobacion_calc) - c.tasa_aprobacion_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_rechazo_informada, c.tasa_rechazo_calc) - c.tasa_rechazo_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_error_informada, c.tasa_error_calc) - c.tasa_error_calc) > c.tolerancia_tasa) WHEN 'SIN_PREDICCION' THEN c.prediccion_id IS NULL END AS dispara FROM c CROSS JOIN alerta_reglas r WHERE r.activa = 1 AND r.regla <> 'VENTANA_FALTANTE' ) SELECT CASE e.regla WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN 'IMPORTE_BAJO' ELSE 'IMPORTE_ALTO' END ELSE e.regla END AS tipo_alerta, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN CASE WHEN e.cantidad_total = 0 AND e.cantidad_total_esperada >= e.volumen_minimo THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= e.corte_pct_critica THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= e.corte_pct_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'VOLUMEN_ALTO' THEN CASE WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= e.corte_pct_critica THEN 'CRITICA' WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= e.corte_pct_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'APROBACION_BAJA' THEN CASE WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= e.corte_tasa_critica THEN 'CRITICA' WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= e.corte_tasa_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'RECHAZO_ALTO' THEN CASE WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= e.corte_tasa_critica THEN 'CRITICA' WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= e.corte_tasa_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'ERROR_ALTO' THEN CASE WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= e.corte_tasa_critica THEN 'CRITICA' WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= e.corte_tasa_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= e.corte_pct_critica THEN 'CRITICA' WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= e.corte_pct_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'LATENCIA_ALTA' THEN CASE WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= e.corte_pct_critica THEN 'CRITICA' WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= e.corte_pct_alta THEN 'ALTA' ELSE 'MEDIA' END WHEN 'SIN_EXITOS' THEN 'CRITICA' WHEN 'DATO_TARDIO' THEN CASE WHEN e.demora_minutos >= 4 * e.gracia_minutos THEN 'ALTA' ELSE 'MEDIA' END WHEN 'DATO_INCONSISTENTE' THEN 'MEDIA' WHEN 'SIN_PREDICCION' THEN 'BAJA' END AS severidad, e.ambiente, e.cliente_id, e.servicio_id, e.producto_id, e.canal_id, e.moneda_id, e.ventana_inicio, e.duracion_ventana_minutos, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN 'cantidad_total' WHEN 'VOLUMEN_ALTO' THEN 'cantidad_total' WHEN 'APROBACION_BAJA' THEN 'tasa_aprobacion' WHEN 'RECHAZO_ALTO' THEN 'tasa_rechazo' WHEN 'ERROR_ALTO' THEN 'tasa_error' WHEN 'IMPORTE_FUERA_DE_BANDA' THEN 'importe_total' WHEN 'LATENCIA_ALTA' THEN 'tiempo_respuesta_promedio' WHEN 'SIN_EXITOS' THEN 'cantidad_exitos' WHEN 'DATO_TARDIO' THEN 'demora_recepcion_minutos' WHEN 'DATO_INCONSISTENTE' THEN 'tasa_aprobacion' WHEN 'SIN_PREDICCION' THEN 'cantidad_total' END AS metrica, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_promedio WHEN 'SIN_EXITOS' THEN e.cantidad_exitos WHEN 'DATO_TARDIO' THEN ROUND(e.demora_minutos, 1) WHEN 'DATO_INCONSISTENTE' THEN e.tasa_aprobacion_informada WHEN 'SIN_PREDICCION' THEN e.cantidad_total END AS valor_observado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_esperada WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_esperada WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_esperada WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_esperada WHEN 'ERROR_ALTO' THEN e.tasa_error_esperada WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total_esperado WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_esperado WHEN 'SIN_EXITOS' THEN e.cantidad_total_esperada * e.tasa_aprobacion_esperada WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_calc, 4) END AS valor_esperado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN e.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN e.importe_total_minimo ELSE e.importe_total_maximo END WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_maximo_esp WHEN 'DATO_TARDIO' THEN e.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN e.tolerancia_tasa END AS umbral, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'VOLUMEN_ALTO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc - e.tasa_aprobacion_esperada, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc - e.tasa_rechazo_esperada, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc - e.tasa_error_esperada, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN ROUND((e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0), 4) WHEN 'LATENCIA_ALTA' THEN ROUND((e.tiempo_respuesta_promedio - e.tiempo_respuesta_esperado) / NULLIF(e.tiempo_respuesta_esperado, 0), 4) WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_informada - e.tasa_aprobacion_calc, 4) END AS desvio, e.metrica_id, e.prediccion_id FROM e WHERE e.dispara = 1;

--- 32. Crear v_alertas_faltantes ---
CREATE VIEW v_alertas_faltantes AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now'), '-' || (horizonte_horas + 1) || ' hours') AS desde_txt, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now')) AS hasta_txt FROM alerta_parametros WHERE id = 1 ) SELECT 'VENTANA_FALTANTE' AS tipo_alerta, CASE WHEN p.cantidad_total_esperada >= COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) THEN 'CRITICA' ELSE 'MEDIA' END AS severidad, p.ambiente, p.cliente_id, p.servicio_id, p.producto_id, p.canal_id, p.moneda_id, p.ventana_inicio, p.duracion_ventana_minutos, 'cantidad_total' AS metrica, NULL AS valor_observado, p.cantidad_total_esperada AS valor_esperado, CASE WHEN COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) = 1 OR p.cantidad_total_esperada IS NULL THEN p.cantidad_total_minima ELSE p.cantidad_total_esperada + COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) * (p.cantidad_total_minima - p.cantidad_total_esperada) END AS umbral, -1.0 AS desvio, NULL AS metrica_id, p.prediccion_id FROM prm CROSS JOIN core_metricas_esperadas p LEFT JOIN umbrales_config ug ON ug.cliente_id = '*' AND ug.servicio_id = '*' LEFT JOIN umbrales_config uc ON uc.cliente_id = p.cliente_id AND uc.servicio_id = '*' LEFT JOIN umbrales_config us ON us.cliente_id = p.cliente_id AND us.servicio_id = p.servicio_id LEFT JOIN feriados fe ON fe.fecha = date(p.ventana_inicio, '-3 hours') WHERE NOT (fe.fecha IS NOT NULL AND COALESCE(us.silenciar_feriados, uc.silenciar_feriados, ug.silenciar_feriados, 0) = 1) AND p.ventana_inicio >= prm.desde_txt AND p.ventana_inicio < prm.hasta_txt AND p.estado_prediccion = 'ACTIVA' AND NOT EXISTS ( SELECT 1 FROM core_metricas_esperadas p2 WHERE p2.ventana_inicio = p.ventana_inicio AND p2.duracion_ventana_minutos = p.duracion_ventana_minutos AND p2.ambiente = p.ambiente AND p2.cliente_id = p.cliente_id AND p2.servicio_id = p.servicio_id AND p2.producto_id = p.producto_id AND p2.canal_id = p.canal_id AND p2.moneda_id = p.moneda_id AND p2.estado_prediccion = 'ACTIVA' AND ( julianday(p2.fecha_generacion) > julianday(p.fecha_generacion) OR ( julianday(p2.fecha_generacion) = julianday(p.fecha_generacion) AND p2.modelo_version > p.modelo_version ) ) ) AND EXISTS (SELECT 1 FROM alerta_reglas WHERE regla = 'VENTANA_FALTANTE' AND activa = 1) AND julianday(p.ventana_fin) + COALESCE(us.gracia_minutos, uc.gracia_minutos, ug.gracia_minutos, prm.gracia_minutos) / 1440.0 <= prm.ahora_jd AND julianday(p.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND p.cantidad_total_esperada > 0 AND NOT EXISTS ( SELECT 1 FROM core_metricas_observadas o WHERE strftime('%Y-%m-%dT%H:%M:%S', o.ventana_inicio) = strftime('%Y-%m-%dT%H:%M:%S', p.ventana_inicio) AND o.duracion_ventana_minutos = p.duracion_ventana_minutos AND o.ambiente = p.ambiente AND o.cliente_id = p.cliente_id AND o.servicio_id = p.servicio_id AND o.producto_id = p.producto_id AND o.canal_id = p.canal_id AND o.moneda_id = p.moneda_id );

--- 33. Crear v_alertas ---
CREATE VIEW v_alertas AS SELECT * FROM v_alertas_observadas UNION ALL SELECT * FROM v_alertas_faltantes;

--- 34. Crear v_prealertas ---
CREATE VIEW v_prealertas AS WITH prm AS ( SELECT volumen_minimo, julianday(COALESCE(ahora, 'now')) AS ahora_jd FROM alerta_parametros WHERE id = 1 ), c AS ( SELECT o.ambiente, o.cliente_id, o.servicio_id, o.producto_id, o.canal_id, o.moneda_id, o.duracion_ventana_minutos, strftime('%Y-%m-%dT%H:%M:%SZ', o.ventana_inicio) AS ventana_inicio, strftime('%Y-%m-%dT%H:%M:%SZ', o.ventana_fin) AS ventana_fin, o.cantidad_total, 1.0 * o.cantidad_exitos / NULLIF(o.cantidad_total, 0) AS aprob, 1.0 * o.cantidad_rechazos / NULLIF(o.cantidad_total, 0) AS rech, 1.0 * o.cantidad_errores / NULLIF(o.cantidad_total, 0) AS err, o.tiempo_respuesta_promedio AS lat, p.cantidad_total_esperada, CASE WHEN COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) = 1 OR p.cantidad_total_esperada IS NULL THEN p.cantidad_total_minima ELSE p.cantidad_total_esperada + COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) * (p.cantidad_total_minima - p.cantidad_total_esperada) END AS cantidad_total_minima, p.tasa_aprobacion_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR p.tasa_aprobacion_esperada IS NULL THEN p.tasa_aprobacion_minima ELSE p.tasa_aprobacion_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_aprobacion_minima - p.tasa_aprobacion_esperada) END AS tasa_aprobacion_minima, p.tasa_rechazo_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR p.tasa_rechazo_esperada IS NULL THEN p.tasa_rechazo_maxima ELSE p.tasa_rechazo_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_rechazo_maxima - p.tasa_rechazo_esperada) END AS tasa_rechazo_maxima, p.tasa_error_esperada, CASE WHEN COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) = 1 OR p.tasa_error_esperada IS NULL THEN p.tasa_error_maxima ELSE p.tasa_error_esperada + COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_error_maxima - p.tasa_error_esperada) END AS tasa_error_maxima, p.tiempo_respuesta_esperado, CASE WHEN COALESCE(us.banda_latencia, uc.banda_latencia, ug.banda_latencia, 1.0) = 1 OR p.tiempo_respuesta_esperado IS NULL THEN p.tiempo_respuesta_maximo ELSE p.tiempo_respuesta_esperado + COALESCE(us.banda_latencia, uc.banda_latencia, ug.banda_latencia, 1.0) * (p.tiempo_respuesta_maximo - p.tiempo_respuesta_esperado) END AS tiempo_respuesta_maximo, (p.cantidad_total_esperada - o.cantidad_total) / NULLIF(COALESCE(us.banda_volumen, uc.banda_volumen, ug.banda_volumen, 1.0) * (p.cantidad_total_esperada - p.cantidad_total_minima), 0) AS m_vol, CASE WHEN o.cantidad_total >= COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) THEN (p.tasa_aprobacion_esperada - 1.0 * o.cantidad_exitos / o.cantidad_total) / NULLIF(COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_aprobacion_esperada - p.tasa_aprobacion_minima), 0) END AS m_aprob, CASE WHEN o.cantidad_total >= COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) THEN (1.0 * o.cantidad_rechazos / o.cantidad_total - p.tasa_rechazo_esperada) / NULLIF(COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_rechazo_maxima - p.tasa_rechazo_esperada), 0) END AS m_rech, CASE WHEN o.cantidad_total >= COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) THEN (1.0 * o.cantidad_errores / o.cantidad_total - p.tasa_error_esperada) / NULLIF(COALESCE(us.banda_tasas, uc.banda_tasas, ug.banda_tasas, 1.0) * (p.tasa_error_maxima - p.tasa_error_esperada), 0) END AS m_err, CASE WHEN o.cantidad_total >= COALESCE(us.volumen_minimo, uc.volumen_minimo, ug.volumen_minimo, prm.volumen_minimo) THEN (o.tiempo_respuesta_promedio - p.tiempo_respuesta_esperado) / NULLIF(COALESCE(us.banda_latencia, uc.banda_latencia, ug.banda_latencia, 1.0) * (p.tiempo_respuesta_maximo - p.tiempo_respuesta_esperado), 0) END AS m_lat FROM core_metricas_observadas o CROSS JOIN core_metricas_esperadas p CROSS JOIN prm LEFT JOIN umbrales_config ug ON ug.cliente_id = '*' AND ug.servicio_id = '*' LEFT JOIN umbrales_config uc ON uc.cliente_id = o.cliente_id AND uc.servicio_id = '*' LEFT JOIN umbrales_config us ON us.cliente_id = o.cliente_id AND us.servicio_id = o.servicio_id LEFT JOIN feriados fe ON fe.fecha = date(o.ventana_inicio, '-3 hours') WHERE NOT (fe.fecha IS NOT NULL AND COALESCE(us.silenciar_feriados, uc.silenciar_feriados, ug.silenciar_feriados, 0) = 1) AND p.ventana_inicio = strftime('%Y-%m-%dT%H:%M:%fZ', o.ventana_inicio) AND p.duracion_ventana_minutos = o.duracion_ventana_minutos AND p.ambiente = o.ambiente AND p.cliente_id = o.cliente_id AND p.servicio_id = o.servicio_id AND p.producto_id = o.producto_id AND p.canal_id = o.canal_id AND p.moneda_id = o.moneda_id AND p.estado_prediccion = 'ACTIVA' AND NOT EXISTS ( SELECT 1 FROM core_metricas_esperadas p2 WHERE p2.ventana_inicio = p.ventana_inicio AND p2.duracion_ventana_minutos = p.duracion_ventana_minutos AND p2.ambiente = p.ambiente AND p2.cliente_id = p.cliente_id AND p2.servicio_id = p.servicio_id AND p2.producto_id = p.producto_id AND p2.canal_id = p.canal_id AND p2.moneda_id = p.moneda_id AND p2.estado_prediccion = 'ACTIVA' AND ( julianday(p2.fecha_generacion) > julianday(p.fecha_generacion) OR ( julianday(p2.fecha_generacion) = julianday(p.fecha_generacion) AND p2.modelo_version > p.modelo_version ) ) ) AND o.ventana_inicio >= (SELECT strftime('%Y-%m-%dT%H:%M:%S', COALESCE(ahora, 'now'), '-49 hours') FROM alerta_parametros WHERE id = 1) AND julianday(o.ventana_fin) <= prm.ahora_jd AND o.cantidad_total > 0 ), t AS ( SELECT c.*, LAG(c.ventana_inicio, 2) OVER s AS ventana_inicio_2, MAX(c.ventana_inicio) OVER (PARTITION BY c.ambiente, c.cliente_id, c.servicio_id, c.producto_id, c.canal_id, c.moneda_id, c.duracion_ventana_minutos) AS ultima_ventana_serie, LAG(c.m_vol, 1) OVER s AS m_vol_1, LAG(c.m_vol, 2) OVER s AS m_vol_2, LAG(c.m_aprob, 1) OVER s AS m_aprob_1, LAG(c.m_aprob, 2) OVER s AS m_aprob_2, LAG(c.m_rech, 1) OVER s AS m_rech_1, LAG(c.m_rech, 2) OVER s AS m_rech_2, LAG(c.m_err, 1) OVER s AS m_err_1, LAG(c.m_err, 2) OVER s AS m_err_2, LAG(c.m_lat, 1) OVER s AS m_lat_1, LAG(c.m_lat, 2) OVER s AS m_lat_2 FROM c WINDOW s AS (PARTITION BY c.ambiente, c.cliente_id, c.servicio_id, c.producto_id, c.canal_id, c.moneda_id, c.duracion_ventana_minutos ORDER BY c.ventana_inicio) ), x AS ( SELECT t.*, k.column1 AS tipo, k.column2 AS tipo_previsto, k.column3 AS metrica, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err ELSE t.m_lat END AS m, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol_1 WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob_1 WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech_1 WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err_1 ELSE t.m_lat_1 END AS m1, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol_2 WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob_2 WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech_2 WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err_2 ELSE t.m_lat_2 END AS m2, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total WHEN 'RIESGO_APROBACION_BAJA' THEN ROUND(t.aprob, 4) WHEN 'RIESGO_RECHAZO_ALTO' THEN ROUND(t.rech, 4) WHEN 'RIESGO_ERROR_ALTO' THEN ROUND(t.err, 4) ELSE t.lat END AS valor_observado, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total_esperada WHEN 'RIESGO_APROBACION_BAJA' THEN t.tasa_aprobacion_esperada WHEN 'RIESGO_RECHAZO_ALTO' THEN t.tasa_rechazo_esperada WHEN 'RIESGO_ERROR_ALTO' THEN t.tasa_error_esperada ELSE t.tiempo_respuesta_esperado END AS valor_esperado, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total_minima WHEN 'RIESGO_APROBACION_BAJA' THEN t.tasa_aprobacion_minima WHEN 'RIESGO_RECHAZO_ALTO' THEN t.tasa_rechazo_maxima WHEN 'RIESGO_ERROR_ALTO' THEN t.tasa_error_maxima ELSE t.tiempo_respuesta_maximo END AS umbral FROM t CROSS JOIN (VALUES ('RIESGO_VOLUMEN_BAJO', 'VOLUMEN_BAJO', 'cantidad_total'), ('RIESGO_APROBACION_BAJA', 'APROBACION_BAJA', 'tasa_aprobacion'), ('RIESGO_RECHAZO_ALTO', 'RECHAZO_ALTO', 'tasa_rechazo'), ('RIESGO_ERROR_ALTO', 'ERROR_ALTO', 'tasa_error'), ('RIESGO_LATENCIA_ALTA', 'LATENCIA_ALTA', 'tiempo_respuesta_promedio') ) AS k ), senal AS ( SELECT x.*, CASE WHEN x.m + (x.m - x.m2) / 2 >= 1 THEN 'ALTA' ELSE 'MEDIA' END AS severidad, CAST(strftime('%s', x.ventana_inicio) AS INTEGER) / (60 * x.duracion_ventana_minutos) - ROW_NUMBER() OVER (PARTITION BY x.ambiente, x.cliente_id, x.servicio_id, x.producto_id, x.canal_id, x.moneda_id, x.duracion_ventana_minutos, x.tipo ORDER BY x.ventana_inicio) AS grupo FROM x WHERE ROUND((julianday(x.ventana_inicio) - julianday(x.ventana_inicio_2)) * 1440) = 2 * x.duracion_ventana_minutos AND COALESCE(x.m_vol, 0) < 1 AND COALESCE(x.m_aprob, 0) < 1 AND COALESCE(x.m_rech, 0) < 1 AND COALESCE(x.m_err, 0) < 1 AND COALESCE(x.m_lat, 0) < 1 AND x.m >= 0.5 AND x.m - x.m1 >= 0.1 AND x.m1 - x.m2 >= 0.1 AND x.m + (x.m - x.m2) >= 1 ), ep AS ( SELECT s.*, MIN(s.ventana_inicio) OVER g AS inicio, COUNT(*) OVER g AS ventanas, MAX(s.m) OVER g AS margen_max, ROW_NUMBER() OVER (PARTITION BY s.ambiente, s.cliente_id, s.servicio_id, s.producto_id, s.canal_id, s.moneda_id, s.duracion_ventana_minutos, s.tipo, s.grupo ORDER BY s.ventana_inicio DESC) AS desde_el_final FROM senal s WINDOW g AS (PARTITION BY s.ambiente, s.cliente_id, s.servicio_id, s.producto_id, s.canal_id, s.moneda_id, s.duracion_ventana_minutos, s.tipo, s.grupo) ), cruce AS ( SELECT ep.*, (SELECT MIN(a.ventana_fin) FROM alertas_historico a WHERE a.ambiente = ep.ambiente AND a.cliente_id = ep.cliente_id AND a.servicio_id = ep.servicio_id AND a.producto_id = ep.producto_id AND a.canal_id = ep.canal_id AND a.moneda_id = ep.moneda_id AND a.duracion_ventana_minutos = ep.duracion_ventana_minutos AND a.tipo_alerta = ep.tipo_previsto AND a.ventana_inicio > ep.inicio AND julianday(a.ventana_inicio) <= julianday(ep.ventana_fin) + 60 / 1440.0 ) AS alerta_real_ventana_fin FROM ep WHERE ep.desde_el_final = 1 ) SELECT cruce.tipo AS tipo_prealerta, cruce.tipo_previsto, cruce.ambiente, cruce.cliente_id, cruce.servicio_id, cruce.producto_id, cruce.canal_id, cruce.moneda_id, cruce.duracion_ventana_minutos, cruce.metrica, cruce.inicio, cruce.ventana_inicio AS ultima_ventana, cruce.ventanas, cruce.severidad, cruce.valor_observado, cruce.valor_esperado, cruce.umbral, ROUND(cruce.m, 2) AS margen_consumido, ROUND(cruce.margen_max, 2) AS margen_consumido_max, CASE cruce.severidad WHEN 'ALTA' THEN 1 ELSE 2 END * cruce.duracion_ventana_minutos AS minutos_estimados, CASE WHEN cruce.ventana_inicio = cruce.ultima_ventana_serie AND cruce.alerta_real_ventana_fin IS NULL AND julianday(cruce.ventana_fin) >= prm.ahora_jd - 3 * cruce.duracion_ventana_minutos / 1440.0 THEN 1 ELSE 0 END AS activa, CASE WHEN cruce.alerta_real_ventana_fin IS NOT NULL THEN 'CONFIRMADA' WHEN prm.ahora_jd < julianday(cruce.ventana_fin) + 60 / 1440.0 THEN 'EN_OBSERVACION' ELSE 'NO_SE_CUMPLIO' END AS resultado, cruce.alerta_real_ventana_fin, ROUND((julianday(cruce.alerta_real_ventana_fin) - julianday(cruce.inicio) - cruce.duracion_ventana_minutos / 1440.0) * 1440) AS minutos_anticipacion FROM cruce CROSS JOIN prm;

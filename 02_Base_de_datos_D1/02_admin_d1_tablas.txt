ETAPA 2 - TABLAS NUEVAS PARA LA CONSOLE DE D1 (equipos y reglas personalizadas)

En el panel de Cloudflare: D1 > helios-monitoring-dev > Console.
Ejecutar los 5 bloques en orden (cada uno es una sola línea). Hacerlo ANTES de publicar el Worker nuevo.
No borra ni modifica nada existente: solo crea tablas e índices nuevos (se puede correr dos veces sin problema).

--- 1. Equipos ---
CREATE TABLE IF NOT EXISTS equipos ( id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL UNIQUE COLLATE NOCASE, descripcion TEXT, responsable_id INTEGER REFERENCES usuarios (id), activo INTEGER NOT NULL DEFAULT 1 CHECK (activo IN (0, 1)), creado TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')) );

--- 2. Miembros de cada equipo ---
CREATE TABLE IF NOT EXISTS equipo_miembros ( equipo_id INTEGER NOT NULL REFERENCES equipos (id), usuario_id INTEGER NOT NULL REFERENCES usuarios (id), PRIMARY KEY (equipo_id, usuario_id) );

--- 3. Índice de miembros por usuario ---
CREATE INDEX IF NOT EXISTS ix_equipo_miembros_usuario ON equipo_miembros (usuario_id);

--- 4. Reglas personalizadas ---
CREATE TABLE IF NOT EXISTS alerta_reglas_personalizadas ( id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL, metrica TEXT NOT NULL CHECK (metrica IN ('VOLUMEN', 'APROBACION', 'RECHAZO', 'ERROR', 'IMPORTE', 'LATENCIA')), direccion TEXT NOT NULL CHECK (direccion IN ('BAJO', 'ALTO')), ventanas_seguidas INTEGER NOT NULL DEFAULT 1 CHECK (ventanas_seguidas BETWEEN 1 AND 8), cliente_id TEXT, servicio_id TEXT, canal_id TEXT, severidad TEXT NOT NULL CHECK (severidad IN ('CRITICA', 'ALTA', 'MEDIA')), activa INTEGER NOT NULL DEFAULT 1 CHECK (activa IN (0, 1)), creada_por INTEGER REFERENCES usuarios (id), creada TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')), actualizada TEXT );

--- 5. Índice de auditoría por usuario ---
CREATE INDEX IF NOT EXISTS ix_auditoria_usuario ON auditoria (usuario_id, fecha);

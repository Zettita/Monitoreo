--- 1. Tabla incidentes_mayores (cada sala de crisis) ---
CREATE TABLE IF NOT EXISTS incidentes_mayores ( id INTEGER PRIMARY KEY AUTOINCREMENT, titulo TEXT NOT NULL, descripcion TEXT, estado TEXT NOT NULL DEFAULT 'ABIERTO' CHECK (estado IN ('ABIERTO', 'MITIGADO', 'RESUELTO', 'CERRADO')), severidad TEXT NOT NULL CHECK (severidad IN ('CRITICA', 'ALTA')), cliente_id TEXT, servicio_id TEXT, responsable_usuario_id INTEGER REFERENCES usuarios (id), abierto_por INTEGER REFERENCES usuarios (id), inicio TEXT NOT NULL, abierto TEXT NOT NULL, mitigado TEXT, resuelto TEXT, cerrado TEXT, causa_raiz TEXT, componente_causa_id INTEGER, resumen_cierre TEXT, origen TEXT NOT NULL DEFAULT 'HELIOS' CHECK (origen IN ('HELIOS', 'IMPORTADO')), actualizado TEXT NOT NULL );

--- 2. Índice de incidentes por estado (lista de salas abiertas y contador del menú) ---
CREATE INDEX IF NOT EXISTS ix_incidentes_mayores_estado ON incidentes_mayores (estado, abierto);

--- 3. Índice de incidentes por componente de la causa (antecedentes de la guía de diagnóstico) ---
CREATE INDEX IF NOT EXISTS ix_incidentes_mayores_componente ON incidentes_mayores (componente_causa_id, cerrado);

--- 4. Tabla incidente_alertas (qué grupos de alertas agrupa cada incidente) ---
CREATE TABLE IF NOT EXISTS incidente_alertas ( incidente_id INTEGER NOT NULL REFERENCES incidentes_mayores (id), alerta_id TEXT NOT NULL, tipo_alerta TEXT NOT NULL, cliente_id TEXT, servicio_id TEXT, canal_id TEXT, sumada_por INTEGER REFERENCES usuarios (id), sumada TEXT NOT NULL, PRIMARY KEY (incidente_id, alerta_id) ) WITHOUT ROWID;

--- 5. Índice de alertas de incidentes por alerta ("es parte de IM-N" en el detalle de la alerta) ---
CREATE INDEX IF NOT EXISTS ix_incidente_alertas_alerta ON incidente_alertas (alerta_id);

--- 6. Tabla incidente_tareas (quién hace qué en la sala) ---
CREATE TABLE IF NOT EXISTS incidente_tareas ( id INTEGER PRIMARY KEY AUTOINCREMENT, incidente_id INTEGER NOT NULL REFERENCES incidentes_mayores (id), titulo TEXT NOT NULL, detalle TEXT, responsable_usuario_id INTEGER REFERENCES usuarios (id), equipo_id INTEGER REFERENCES equipos (id), componente_id INTEGER, estado TEXT NOT NULL DEFAULT 'PENDIENTE' CHECK (estado IN ('PENDIENTE', 'EN_CURSO', 'HECHA', 'CANCELADA')), resultado TEXT, creada_por INTEGER REFERENCES usuarios (id), creada TEXT NOT NULL, vence TEXT, actualizada TEXT NOT NULL, terminada TEXT, propuesta_por_asistente INTEGER NOT NULL DEFAULT 0 );

--- 7. Índice de tareas por incidente y estado ---
CREATE INDEX IF NOT EXISTS ix_incidente_tareas_incidente ON incidente_tareas (incidente_id, estado);

--- 8. Índice de tareas por responsable y estado ("mis tareas" de Soporte) ---
CREATE INDEX IF NOT EXISTS ix_incidente_tareas_responsable ON incidente_tareas (responsable_usuario_id, estado);

--- 9. Tabla incidente_linea_tiempo (cada hecho del incidente con su hora) ---
CREATE TABLE IF NOT EXISTS incidente_linea_tiempo ( id INTEGER PRIMARY KEY AUTOINCREMENT, incidente_id INTEGER NOT NULL REFERENCES incidentes_mayores (id), fecha TEXT NOT NULL, usuario_id INTEGER REFERENCES usuarios (id), tipo TEXT NOT NULL CHECK (tipo IN ('APERTURA', 'ALERTA_SUMADA', 'ESTADO', 'TAREA', 'NOTA', 'COMUNICADO', 'ESCALAMIENTO', 'CIERRE')), texto TEXT NOT NULL, referencia TEXT );

--- 10. Índice de la línea de tiempo por incidente y fecha ---
CREATE INDEX IF NOT EXISTS ix_incidente_linea_tiempo_incidente ON incidente_linea_tiempo (incidente_id, fecha);

--- 11. Tabla componentes_infraestructura (catálogo para la guía de diagnóstico) ---
CREATE TABLE IF NOT EXISTS componentes_infraestructura ( id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL UNIQUE, tipo TEXT NOT NULL CHECK (tipo IN ('GATEWAY', 'BASE_DE_DATOS', 'AUTENTICACION', 'PROVEEDOR_EXTERNO', 'RED', 'APLICACION')), descripcion TEXT, area_responsable TEXT, equipo_id INTEGER REFERENCES equipos (id), que_revisar TEXT, activo INTEGER NOT NULL DEFAULT 1 );

--- 12. Tabla componente_dependencias (de qué depende cada servicio y cada componente) ---
CREATE TABLE IF NOT EXISTS componente_dependencias ( id INTEGER PRIMARY KEY AUTOINCREMENT, cliente_id TEXT, servicio_id TEXT, componente_id INTEGER REFERENCES componentes_infraestructura (id), depende_de_componente_id INTEGER NOT NULL REFERENCES componentes_infraestructura (id), CHECK ((servicio_id IS NULL) <> (componente_id IS NULL)) );

--- 13. Índice único de dependencias (repetir la carga no duplica) ---
CREATE UNIQUE INDEX IF NOT EXISTS ux_componente_dependencias ON componente_dependencias (IFNULL(cliente_id, '*'), IFNULL(servicio_id, ''), IFNULL(componente_id, 0), depende_de_componente_id);

--- 14. Índice de dependencias por servicio (la guía arranca por el servicio afectado) ---
CREATE INDEX IF NOT EXISTS ix_componente_dependencias_servicio ON componente_dependencias (servicio_id, cliente_id);

Qué cambia: las vistas dejan de recorrer las tablas enteras de métricas y predicciones en cada revisión de alertas y en cada carga del tablero.
Ahora buscan solo las ventanas de las últimas 25 horas por el índice de ventana_inicio, así que lo que leen ya no crece con la historia.
Devuelven exactamente lo mismo: probado comparando vista vieja contra nueva en 209 momentos de la demo, con predicciones nuevas e inactivas agregadas, y con la API de Power BI.

En el panel de Cloudflare: D1 > helios-monitoring-dev > Console. Cada bloque es una sola línea.
Correr el bloque 0 primero: tiene que dar 0 y 0. Si da otro número, NO seguir y avisar (hay fechas guardadas con otro formato).
Después correr los bloques 1 a 8 en orden y seguidos: entre el 1 y el 8 el tablero y el proceso de alertas dan error porque faltan las vistas.

--- 0. Verificación del formato de fechas (tiene que dar 0 y 0) ---
SELECT (SELECT COUNT(*) FROM core_metricas_observadas WHERE ventana_inicio <> strftime('%Y-%m-%dT%H:%M:%fZ', ventana_inicio) OR ventana_fin <> strftime('%Y-%m-%dT%H:%M:%fZ', ventana_fin)) AS observadas_con_otro_formato, (SELECT COUNT(*) FROM core_metricas_esperadas WHERE ventana_inicio <> strftime('%Y-%m-%dT%H:%M:%fZ', ventana_inicio) OR ventana_fin <> strftime('%Y-%m-%dT%H:%M:%fZ', ventana_fin)) AS esperadas_con_otro_formato;

--- 5. Crear v_comparacion ---
CREATE VIEW v_comparacion AS SELECT o.metrica_id, p.prediccion_id, o.ventana_inicio, o.ventana_fin, o.duracion_ventana_minutos, o.ambiente, o.cliente_id, o.servicio_id, o.producto_id, o.canal_id, o.moneda_id, o.cantidad_total, o.cantidad_exitos, o.cantidad_rechazos, o.cantidad_errores, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_exitos / o.cantidad_total END AS tasa_aprobacion_calc, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_rechazos / o.cantidad_total END AS tasa_rechazo_calc, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_errores / o.cantidad_total END AS tasa_error_calc, o.tasa_aprobacion AS tasa_aprobacion_informada, o.tasa_rechazo AS tasa_rechazo_informada, o.tasa_error AS tasa_error_informada, o.importe_total, o.tiempo_respuesta_promedio, o.tiempo_respuesta_maximo AS tiempo_respuesta_maximo_obs, o.ultimo_evento_exitoso, o.fecha_recepcion, o.origen_dato, p.cantidad_total_esperada, p.cantidad_total_minima, p.cantidad_total_maxima, p.tasa_aprobacion_esperada, p.tasa_aprobacion_minima, p.tasa_rechazo_esperada, p.tasa_rechazo_maxima, p.tasa_error_esperada, p.tasa_error_maxima, p.importe_total_esperado, p.importe_total_minimo, p.importe_total_maximo, p.tiempo_respuesta_esperado, p.tiempo_respuesta_maximo AS tiempo_respuesta_maximo_esp, p.modelo_id, p.modelo_version FROM core_metricas_observadas o LEFT JOIN core_metricas_esperadas p ON p.ventana_inicio = strftime('%Y-%m-%dT%H:%M:%fZ', o.ventana_inicio) AND p.duracion_ventana_minutos = o.duracion_ventana_minutos AND p.ambiente = o.ambiente AND p.cliente_id = o.cliente_id AND p.servicio_id = o.servicio_id AND p.producto_id = o.producto_id AND p.canal_id = o.canal_id AND p.moneda_id = o.moneda_id AND p.estado_prediccion = 'ACTIVA' AND NOT EXISTS ( SELECT 1 FROM core_metricas_esperadas p2 WHERE p2.ventana_inicio = p.ventana_inicio AND p2.duracion_ventana_minutos = p.duracion_ventana_minutos AND p2.ambiente = p.ambiente AND p2.cliente_id = p.cliente_id AND p2.servicio_id = p.servicio_id AND p2.producto_id = p.producto_id AND p2.canal_id = p.canal_id AND p2.moneda_id = p.moneda_id AND p2.estado_prediccion = 'ACTIVA' AND ( julianday(p2.fecha_generacion) > julianday(p.fecha_generacion) OR ( julianday(p2.fecha_generacion) = julianday(p.fecha_generacion) AND p2.modelo_version > p.modelo_version ) ) );

--- 6. Crear v_alertas_observadas ---
CREATE VIEW v_alertas_observadas AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, tolerancia_tasa, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now'), '-' || (horizonte_horas + 1) || ' hours') AS desde_txt, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now')) AS hasta_txt FROM alerta_parametros WHERE id = 1 ), c AS ( SELECT v.*, prm.volumen_minimo, prm.gracia_minutos, prm.tolerancia_tasa, (julianday(v.fecha_recepcion) - julianday(v.ventana_fin)) * 1440 AS demora_minutos FROM prm CROSS JOIN v_comparacion v WHERE v.ventana_inicio >= prm.desde_txt AND v.ventana_inicio < prm.hasta_txt AND julianday(v.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND julianday(v.ventana_fin) <= prm.ahora_jd ), e AS ( SELECT c.*, r.regla, CASE r.regla WHEN 'VOLUMEN_BAJO' THEN c.cantidad_total_minima IS NOT NULL AND c.cantidad_total < c.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN c.cantidad_total_maxima IS NOT NULL AND c.cantidad_total > c.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_aprobacion_minima IS NOT NULL AND c.tasa_aprobacion_calc < c.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_rechazo_maxima IS NOT NULL AND c.tasa_rechazo_calc > c.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_error_maxima IS NOT NULL AND c.tasa_error_calc > c.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN (c.importe_total_minimo IS NOT NULL AND c.importe_total < c.importe_total_minimo) OR (c.importe_total_maximo IS NOT NULL AND c.importe_total > c.importe_total_maximo) WHEN 'LATENCIA_ALTA' THEN c.tiempo_respuesta_maximo_esp IS NOT NULL AND c.tiempo_respuesta_promedio > c.tiempo_respuesta_maximo_esp WHEN 'SIN_EXITOS' THEN c.cantidad_total >= c.volumen_minimo AND c.cantidad_exitos = 0 WHEN 'DATO_TARDIO' THEN c.origen_dato = 'API' AND c.demora_minutos > c.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN c.cantidad_total > 0 AND ( ABS(COALESCE(c.tasa_aprobacion_informada, c.tasa_aprobacion_calc) - c.tasa_aprobacion_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_rechazo_informada, c.tasa_rechazo_calc) - c.tasa_rechazo_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_error_informada, c.tasa_error_calc) - c.tasa_error_calc) > c.tolerancia_tasa) WHEN 'SIN_PREDICCION' THEN c.prediccion_id IS NULL END AS dispara FROM c CROSS JOIN alerta_reglas r WHERE r.activa = 1 AND r.regla <> 'VENTANA_FALTANTE' ) SELECT CASE e.regla WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN 'IMPORTE_BAJO' ELSE 'IMPORTE_ALTO' END ELSE e.regla END AS tipo_alerta, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN CASE WHEN e.cantidad_total = 0 AND e.cantidad_total_esperada >= e.volumen_minimo THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= 0.50 THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'VOLUMEN_ALTO' THEN CASE WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= 0.50 THEN 'CRITICA' WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'APROBACION_BAJA' THEN CASE WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= 0.15 THEN 'CRITICA' WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'RECHAZO_ALTO' THEN CASE WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= 0.15 THEN 'CRITICA' WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'ERROR_ALTO' THEN CASE WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= 0.15 THEN 'CRITICA' WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= 0.50 THEN 'CRITICA' WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'LATENCIA_ALTA' THEN CASE WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= 0.50 THEN 'CRITICA' WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'SIN_EXITOS' THEN 'CRITICA' WHEN 'DATO_TARDIO' THEN CASE WHEN e.demora_minutos >= 4 * e.gracia_minutos THEN 'ALTA' ELSE 'MEDIA' END WHEN 'DATO_INCONSISTENTE' THEN 'MEDIA' WHEN 'SIN_PREDICCION' THEN 'BAJA' END AS severidad, e.ambiente, e.cliente_id, e.servicio_id, e.producto_id, e.canal_id, e.moneda_id, e.ventana_inicio, e.duracion_ventana_minutos, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN 'cantidad_total' WHEN 'VOLUMEN_ALTO' THEN 'cantidad_total' WHEN 'APROBACION_BAJA' THEN 'tasa_aprobacion' WHEN 'RECHAZO_ALTO' THEN 'tasa_rechazo' WHEN 'ERROR_ALTO' THEN 'tasa_error' WHEN 'IMPORTE_FUERA_DE_BANDA' THEN 'importe_total' WHEN 'LATENCIA_ALTA' THEN 'tiempo_respuesta_promedio' WHEN 'SIN_EXITOS' THEN 'cantidad_exitos' WHEN 'DATO_TARDIO' THEN 'demora_recepcion_minutos' WHEN 'DATO_INCONSISTENTE' THEN 'tasa_aprobacion' WHEN 'SIN_PREDICCION' THEN 'cantidad_total' END AS metrica, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_promedio WHEN 'SIN_EXITOS' THEN e.cantidad_exitos WHEN 'DATO_TARDIO' THEN ROUND(e.demora_minutos, 1) WHEN 'DATO_INCONSISTENTE' THEN e.tasa_aprobacion_informada WHEN 'SIN_PREDICCION' THEN e.cantidad_total END AS valor_observado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_esperada WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_esperada WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_esperada WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_esperada WHEN 'ERROR_ALTO' THEN e.tasa_error_esperada WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total_esperado WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_esperado WHEN 'SIN_EXITOS' THEN e.cantidad_total_esperada * e.tasa_aprobacion_esperada WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_calc, 4) END AS valor_esperado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN e.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN e.importe_total_minimo ELSE e.importe_total_maximo END WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_maximo_esp WHEN 'DATO_TARDIO' THEN e.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN e.tolerancia_tasa END AS umbral, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'VOLUMEN_ALTO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc - e.tasa_aprobacion_esperada, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc - e.tasa_rechazo_esperada, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc - e.tasa_error_esperada, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN ROUND((e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0), 4) WHEN 'LATENCIA_ALTA' THEN ROUND((e.tiempo_respuesta_promedio - e.tiempo_respuesta_esperado) / NULLIF(e.tiempo_respuesta_esperado, 0), 4) WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_informada - e.tasa_aprobacion_calc, 4) END AS desvio, e.metrica_id, e.prediccion_id FROM e WHERE e.dispara = 1;

--- 7. Crear v_alertas_faltantes ---
CREATE VIEW v_alertas_faltantes AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now'), '-' || (horizonte_horas + 1) || ' hours') AS desde_txt, strftime('%Y-%m-%dT%H:%M:%fZ', COALESCE(ahora, 'now')) AS hasta_txt FROM alerta_parametros WHERE id = 1 ) SELECT 'VENTANA_FALTANTE' AS tipo_alerta, CASE WHEN p.cantidad_total_esperada >= prm.volumen_minimo THEN 'CRITICA' ELSE 'MEDIA' END AS severidad, p.ambiente, p.cliente_id, p.servicio_id, p.producto_id, p.canal_id, p.moneda_id, p.ventana_inicio, p.duracion_ventana_minutos, 'cantidad_total' AS metrica, NULL AS valor_observado, p.cantidad_total_esperada AS valor_esperado, p.cantidad_total_minima AS umbral, -1.0 AS desvio, NULL AS metrica_id, p.prediccion_id FROM prm CROSS JOIN core_metricas_esperadas p WHERE p.ventana_inicio >= prm.desde_txt AND p.ventana_inicio < prm.hasta_txt AND p.estado_prediccion = 'ACTIVA' AND NOT EXISTS ( SELECT 1 FROM core_metricas_esperadas p2 WHERE p2.ventana_inicio = p.ventana_inicio AND p2.duracion_ventana_minutos = p.duracion_ventana_minutos AND p2.ambiente = p.ambiente AND p2.cliente_id = p.cliente_id AND p2.servicio_id = p.servicio_id AND p2.producto_id = p.producto_id AND p2.canal_id = p.canal_id AND p2.moneda_id = p.moneda_id AND p2.estado_prediccion = 'ACTIVA' AND ( julianday(p2.fecha_generacion) > julianday(p.fecha_generacion) OR ( julianday(p2.fecha_generacion) = julianday(p.fecha_generacion) AND p2.modelo_version > p.modelo_version ) ) ) AND EXISTS (SELECT 1 FROM alerta_reglas WHERE regla = 'VENTANA_FALTANTE' AND activa = 1) AND julianday(p.ventana_fin) + prm.gracia_minutos / 1440.0 <= prm.ahora_jd AND julianday(p.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND p.cantidad_total_esperada > 0 AND NOT EXISTS ( SELECT 1 FROM core_metricas_observadas o WHERE strftime('%Y-%m-%dT%H:%M:%S', o.ventana_inicio) = strftime('%Y-%m-%dT%H:%M:%S', p.ventana_inicio) AND o.duracion_ventana_minutos = p.duracion_ventana_minutos AND o.ambiente = p.ambiente AND o.cliente_id = p.cliente_id AND o.servicio_id = p.servicio_id AND o.producto_id = p.producto_id AND o.canal_id = p.canal_id AND o.moneda_id = p.moneda_id );

--- 8. Crear v_alertas ---
CREATE VIEW v_alertas AS SELECT * FROM v_alertas_observadas UNION ALL SELECT * FROM v_alertas_faltantes;

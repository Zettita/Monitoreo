  vacío = técnico).

- Pasa las listas de difusión que ya existían a equipos, con las mismas personas y roles. Si ya había un equipo con el mismo
  nombre, las personas de la lista se suman a ese equipo. No borra alerta_parametros.listas_difusion: queda como respaldo.

--- 1. Rol de cada persona en los comunicados (una columna en equipo_miembros) ---
ALTER TABLE equipo_miembros ADD COLUMN rol_comunicado TEXT CHECK (rol_comunicado IS NULL OR rol_comunicado IN ('INFORMADO', 'TECNICO', 'LIDER'));

--- 2. Para qué alertas sirve el equipo (una columna en equipos) ---
ALTER TABLE equipos ADD COLUMN alcance_cliente_id TEXT;

--- 3. Pasar las listas de difusión a equipos ---
INSERT OR IGNORE INTO equipos (nombre, descripcion, activo, alcance_cliente_id) SELECT json_extract(l.value, '$.nombre'), 'Lista de difusión pasada a equipo', CASE WHEN json_extract(l.value, '$.activa') = 0 THEN 0 ELSE 1 END, json_extract(l.value, '$.cliente_id') FROM alerta_parametros p, json_each(p.listas_difusion, '$.listas') l WHERE p.id = 1 AND json_valid(p.listas_difusion);

--- 4. Pasar las personas de cada lista (con su rol) a su equipo ---
INSERT OR IGNORE INTO equipo_miembros (equipo_id, usuario_id, rol_comunicado) SELECT e.id, json_extract(m.value, '$.usuario_id'), json_extract(m.value, '$.rol') FROM alerta_parametros p, json_each(p.listas_difusion, '$.listas') l, json_each(l.value, '$.miembros') m, equipos e WHERE p.id = 1 AND json_valid(p.listas_difusion) AND e.nombre = json_extract(l.value, '$.nombre') COLLATE NOCASE AND json_extract(m.value, '$.usuario_id') IN (SELECT id FROM usuarios);

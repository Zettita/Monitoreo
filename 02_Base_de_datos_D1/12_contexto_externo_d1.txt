--- 1. Tabla contexto_externo (lo último que se trajo de proveedores y noticias; 1 fila por fuente) ---
CREATE TABLE IF NOT EXISTS contexto_externo ( fuente TEXT PRIMARY KEY, actualizado TEXT NOT NULL, contenido TEXT NOT NULL, error TEXT ) WITHOUT ROWID;

--- 2. Tabla eventos_comerciales (Black Friday, CyberMonday, Hot Sale…; se cargan en Administración > Umbrales) ---
CREATE TABLE IF NOT EXISTS eventos_comerciales ( id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL, fecha_desde TEXT NOT NULL, fecha_hasta TEXT NOT NULL, cliente_id TEXT, servicio_id TEXT, efecto_esperado TEXT NOT NULL DEFAULT 'MAS_TRAFICO' CHECK (efecto_esperado IN ('MAS_TRAFICO', 'MENOS_TRAFICO')), nota TEXT, creado_por INTEGER REFERENCES usuarios (id), creado TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')), CHECK (fecha_hasta >= fecha_desde) );

--- 3. Índice por fechas de los eventos ---
CREATE INDEX IF NOT EXISTS ix_eventos_comerciales_fechas ON eventos_comerciales (fecha_hasta, fecha_desde);

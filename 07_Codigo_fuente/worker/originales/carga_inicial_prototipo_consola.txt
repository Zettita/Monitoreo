--- 15. Histórico: última detección y cantidad de detecciones ---
UPDATE alertas_historico SET ultima_deteccion = CASE WHEN estado = 'RESUELTA' THEN MAX(primera_deteccion, strftime('%Y-%m-%dT%H:%M:%SZ', fecha_resolucion, '-10 minutes')) ELSE strftime('%Y-%m-%dT%H:%M:%SZ', 'now') END, cantidad_detecciones = 1 + CAST((julianday(CASE WHEN estado = 'RESUELTA' THEN fecha_resolucion ELSE strftime('%Y-%m-%dT%H:%M:%SZ', 'now') END) - julianday(primera_deteccion)) * 96 AS INTEGER);

--- 16. Histórico: la guardia reconoció los incidentes críticos y altos ya resueltos ---
UPDATE alertas_historico SET reconocida_por = CASE abs(unicode(substr(cliente_id, 7)) + length(tipo_alerta)) % 3 WHEN 0 THEN 'guardia.noc' WHEN 1 THEN 'm.gomez' ELSE 'l.perez' END, fecha_reconocimiento = MIN(fecha_resolucion, strftime('%Y-%m-%dT%H:%M:%SZ', primera_deteccion, '+8 minutes')) WHERE estado = 'RESUELTA' AND severidad_maxima IN ('CRITICA', 'ALTA');

--- 17. Histórico: incidente de latencia en curso tomado por la guardia ---
UPDATE alertas_historico SET estado = 'RECONOCIDA', reconocida_por = 'm.gomez', fecha_reconocimiento = MIN(strftime('%Y-%m-%dT%H:%M:%SZ', 'now'), strftime('%Y-%m-%dT%H:%M:%SZ', primera_deteccion, '+8 minutes')), comentario = 'Revisando rendimiento del proveedor de recargas' WHERE estado = 'ABIERTA' AND tipo_alerta = 'LATENCIA_ALTA' AND cliente_id = 'CLI_003';

--- 18. Configuración: horizonte normal de 24 horas ---
UPDATE alerta_parametros SET horizonte_horas = 24 WHERE id = 1;

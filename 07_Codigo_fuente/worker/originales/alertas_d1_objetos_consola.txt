OBJETOS DE ALERTAS - VERSIÓN PARA LA CONSOLE DE D1

En el panel de Cloudflare: D1 > helios-monitoring-dev > Console.
Copiar cada bloque (es una sola línea: clic al inicio y Shift + Fin), pegarlo y hacer clic en Execute. Ejecutar en orden, del 1 al último.
No borra métricas ni alertas: recrea las reglas, los parámetros (vuelven a sus valores por defecto: reloj en hora real) y las vistas.
alertas_historico se crea solo si no existe, así que el histórico se conserva.

===== A. OBJETOS (parámetros, reglas, vistas de alertas e índice) - de alertas_d1_objetos.sql =====

--- 1 ---
DROP TABLE IF EXISTS alerta_parametros;

--- 2 ---
CREATE TABLE alerta_parametros ( id INTEGER PRIMARY KEY CHECK (id = 1), ahora TEXT, volumen_minimo INTEGER NOT NULL DEFAULT 20, gracia_minutos INTEGER NOT NULL DEFAULT 5, horizonte_horas INTEGER NOT NULL DEFAULT 24, tolerancia_tasa REAL NOT NULL DEFAULT 0.001 );

--- 3 ---
INSERT INTO alerta_parametros (id) VALUES (1);

--- 4 ---
DROP VIEW IF EXISTS v_prediccion_vigente;

--- 5 ---
CREATE VIEW v_prediccion_vigente AS SELECT * FROM ( SELECT e.*, strftime('%Y-%m-%dT%H:%M:%S', e.ventana_inicio) AS ventana_inicio_norm, ROW_NUMBER() OVER ( PARTITION BY strftime('%Y-%m-%dT%H:%M:%S', e.ventana_inicio), e.duracion_ventana_minutos, e.ambiente, e.cliente_id, e.servicio_id, e.producto_id, e.canal_id, e.moneda_id ORDER BY julianday(e.fecha_generacion) DESC, e.modelo_version DESC ) AS rn FROM core_metricas_esperadas e WHERE e.estado_prediccion = 'ACTIVA' ) WHERE rn = 1;

--- 6 ---
DROP VIEW IF EXISTS v_comparacion;

--- 7 ---
CREATE VIEW v_comparacion AS SELECT o.metrica_id, p.prediccion_id, o.ventana_inicio, o.ventana_fin, o.duracion_ventana_minutos, o.ambiente, o.cliente_id, o.servicio_id, o.producto_id, o.canal_id, o.moneda_id, o.cantidad_total, o.cantidad_exitos, o.cantidad_rechazos, o.cantidad_errores, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_exitos / o.cantidad_total END AS tasa_aprobacion_calc, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_rechazos / o.cantidad_total END AS tasa_rechazo_calc, CASE WHEN o.cantidad_total > 0 THEN 1.0 * o.cantidad_errores / o.cantidad_total END AS tasa_error_calc, o.tasa_aprobacion AS tasa_aprobacion_informada, o.tasa_rechazo AS tasa_rechazo_informada, o.tasa_error AS tasa_error_informada, o.importe_total, o.tiempo_respuesta_promedio, o.tiempo_respuesta_maximo AS tiempo_respuesta_maximo_obs, o.ultimo_evento_exitoso, o.fecha_recepcion, o.origen_dato, p.cantidad_total_esperada, p.cantidad_total_minima, p.cantidad_total_maxima, p.tasa_aprobacion_esperada, p.tasa_aprobacion_minima, p.tasa_rechazo_esperada, p.tasa_rechazo_maxima, p.tasa_error_esperada, p.tasa_error_maxima, p.importe_total_esperado, p.importe_total_minimo, p.importe_total_maximo, p.tiempo_respuesta_esperado, p.tiempo_respuesta_maximo AS tiempo_respuesta_maximo_esp, p.modelo_id, p.modelo_version FROM core_metricas_observadas o LEFT JOIN v_prediccion_vigente p ON p.ventana_inicio_norm = strftime('%Y-%m-%dT%H:%M:%S', o.ventana_inicio) AND p.duracion_ventana_minutos = o.duracion_ventana_minutos AND p.ambiente = o.ambiente AND p.cliente_id = o.cliente_id AND p.servicio_id = o.servicio_id AND p.producto_id = o.producto_id AND p.canal_id = o.canal_id AND p.moneda_id = o.moneda_id;

INSERT INTO alerta_reglas (regla, orden, grupo, descripcion) VALUES ('VOLUMEN_BAJO', 1, 'VOLUMEN', 'Caída de tráfico por debajo del mínimo esperado'), ('VOLUMEN_ALTO', 2, 'VOLUMEN', 'Pico de tráfico por encima del máximo esperado'), ('APROBACION_BAJA', 3, 'TRANSACCIONES', 'Tasa de aprobación por debajo del mínimo'), ('RECHAZO_ALTO', 4, 'TRANSACCIONES', 'Tasa de rechazo por encima del máximo'), ('ERROR_ALTO', 5, 'TRANSACCIONES', 'Tasa de error técnico por encima del máximo'), ('IMPORTE_FUERA_DE_BANDA', 6, 'TRANSACCIONES', 'Importe total fuera de la banda (genera IMPORTE_BAJO o IMPORTE_ALTO)'), ('LATENCIA_ALTA', 7, 'TRANSACCIONES', 'Tiempo de respuesta promedio por encima del máximo'), ('SIN_EXITOS', 8, 'TRANSACCIONES', 'Hay tráfico pero ninguna transacción aprobada'), ('DATO_TARDIO', 9, 'CALIDAD_DATO', 'La métrica llegó después de la gracia'), ('DATO_INCONSISTENTE', 10, 'CALIDAD_DATO', 'Tasa informada distinta de la recalculada'), ('SIN_PREDICCION', 11, 'CALIDAD_DATO', 'Dato observado sin predicción del modelo'), ('VENTANA_FALTANTE', 12, 'CALIDAD_DATO', 'Se esperaba dato y no llegó');

--- 13 ---
CREATE VIEW v_alertas_observadas AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, tolerancia_tasa, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd FROM alerta_parametros WHERE id = 1 ), c AS ( SELECT v.*, prm.volumen_minimo, prm.gracia_minutos, prm.tolerancia_tasa, (julianday(v.fecha_recepcion) - julianday(v.ventana_fin)) * 1440 AS demora_minutos FROM v_comparacion v CROSS JOIN prm WHERE julianday(v.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND julianday(v.ventana_fin) <= prm.ahora_jd ), e AS ( SELECT c.*, r.regla, CASE r.regla WHEN 'VOLUMEN_BAJO' THEN c.cantidad_total_minima IS NOT NULL AND c.cantidad_total < c.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN c.cantidad_total_maxima IS NOT NULL AND c.cantidad_total > c.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_aprobacion_minima IS NOT NULL AND c.tasa_aprobacion_calc < c.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_rechazo_maxima IS NOT NULL AND c.tasa_rechazo_calc > c.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN c.cantidad_total >= c.volumen_minimo AND c.tasa_error_maxima IS NOT NULL AND c.tasa_error_calc > c.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN (c.importe_total_minimo IS NOT NULL AND c.importe_total < c.importe_total_minimo) OR (c.importe_total_maximo IS NOT NULL AND c.importe_total > c.importe_total_maximo) WHEN 'LATENCIA_ALTA' THEN c.tiempo_respuesta_maximo_esp IS NOT NULL AND c.tiempo_respuesta_promedio > c.tiempo_respuesta_maximo_esp WHEN 'SIN_EXITOS' THEN c.cantidad_total >= c.volumen_minimo AND c.cantidad_exitos = 0 WHEN 'DATO_TARDIO' THEN c.origen_dato = 'API' AND c.demora_minutos > c.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN c.cantidad_total > 0 AND ( ABS(COALESCE(c.tasa_aprobacion_informada, c.tasa_aprobacion_calc) - c.tasa_aprobacion_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_rechazo_informada, c.tasa_rechazo_calc) - c.tasa_rechazo_calc) > c.tolerancia_tasa OR ABS(COALESCE(c.tasa_error_informada, c.tasa_error_calc) - c.tasa_error_calc) > c.tolerancia_tasa) WHEN 'SIN_PREDICCION' THEN c.prediccion_id IS NULL END AS dispara FROM c CROSS JOIN alerta_reglas r WHERE r.activa = 1 AND r.regla <> 'VENTANA_FALTANTE' ) SELECT CASE e.regla WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN 'IMPORTE_BAJO' ELSE 'IMPORTE_ALTO' END ELSE e.regla END AS tipo_alerta, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN CASE WHEN e.cantidad_total = 0 AND e.cantidad_total_esperada >= e.volumen_minimo THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= 0.50 THEN 'CRITICA' WHEN (e.cantidad_total_esperada - e.cantidad_total) / NULLIF(e.cantidad_total_esperada, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'VOLUMEN_ALTO' THEN CASE WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= 0.50 THEN 'CRITICA' WHEN (e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'APROBACION_BAJA' THEN CASE WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= 0.15 THEN 'CRITICA' WHEN e.tasa_aprobacion_esperada - e.tasa_aprobacion_calc >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'RECHAZO_ALTO' THEN CASE WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= 0.15 THEN 'CRITICA' WHEN e.tasa_rechazo_calc - COALESCE(e.tasa_rechazo_esperada, e.tasa_rechazo_maxima) >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'ERROR_ALTO' THEN CASE WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= 0.15 THEN 'CRITICA' WHEN e.tasa_error_calc - COALESCE(e.tasa_error_esperada, e.tasa_error_maxima) >= 0.05 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= 0.50 THEN 'CRITICA' WHEN ABS(e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'LATENCIA_ALTA' THEN CASE WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= 0.50 THEN 'CRITICA' WHEN (e.tiempo_respuesta_promedio - COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp)) / NULLIF(COALESCE(e.tiempo_respuesta_esperado, e.tiempo_respuesta_maximo_esp), 0) >= 0.25 THEN 'ALTA' ELSE 'MEDIA' END WHEN 'SIN_EXITOS' THEN 'CRITICA' WHEN 'DATO_TARDIO' THEN CASE WHEN e.demora_minutos >= 4 * e.gracia_minutos THEN 'ALTA' ELSE 'MEDIA' END WHEN 'DATO_INCONSISTENTE' THEN 'MEDIA' WHEN 'SIN_PREDICCION' THEN 'BAJA' END AS severidad, e.ambiente, e.cliente_id, e.servicio_id, e.producto_id, e.canal_id, e.moneda_id, e.ventana_inicio, e.duracion_ventana_minutos, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN 'cantidad_total' WHEN 'VOLUMEN_ALTO' THEN 'cantidad_total' WHEN 'APROBACION_BAJA' THEN 'tasa_aprobacion' WHEN 'RECHAZO_ALTO' THEN 'tasa_rechazo' WHEN 'ERROR_ALTO' THEN 'tasa_error' WHEN 'IMPORTE_FUERA_DE_BANDA' THEN 'importe_total' WHEN 'LATENCIA_ALTA' THEN 'tiempo_respuesta_promedio' WHEN 'SIN_EXITOS' THEN 'cantidad_exitos' WHEN 'DATO_TARDIO' THEN 'demora_recepcion_minutos' WHEN 'DATO_INCONSISTENTE' THEN 'tasa_aprobacion' WHEN 'SIN_PREDICCION' THEN 'cantidad_total' END AS metrica, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_promedio WHEN 'SIN_EXITOS' THEN e.cantidad_exitos WHEN 'DATO_TARDIO' THEN ROUND(e.demora_minutos, 1) WHEN 'DATO_INCONSISTENTE' THEN e.tasa_aprobacion_informada WHEN 'SIN_PREDICCION' THEN e.cantidad_total END AS valor_observado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_esperada WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_esperada WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_esperada WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_esperada WHEN 'ERROR_ALTO' THEN e.tasa_error_esperada WHEN 'IMPORTE_FUERA_DE_BANDA' THEN e.importe_total_esperado WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_esperado WHEN 'SIN_EXITOS' THEN e.cantidad_total_esperada * e.tasa_aprobacion_esperada WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_calc, 4) END AS valor_esperado, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN e.cantidad_total_minima WHEN 'VOLUMEN_ALTO' THEN e.cantidad_total_maxima WHEN 'APROBACION_BAJA' THEN e.tasa_aprobacion_minima WHEN 'RECHAZO_ALTO' THEN e.tasa_rechazo_maxima WHEN 'ERROR_ALTO' THEN e.tasa_error_maxima WHEN 'IMPORTE_FUERA_DE_BANDA' THEN CASE WHEN e.importe_total < e.importe_total_minimo THEN e.importe_total_minimo ELSE e.importe_total_maximo END WHEN 'LATENCIA_ALTA' THEN e.tiempo_respuesta_maximo_esp WHEN 'DATO_TARDIO' THEN e.gracia_minutos WHEN 'DATO_INCONSISTENTE' THEN e.tolerancia_tasa END AS umbral, CASE e.regla WHEN 'VOLUMEN_BAJO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'VOLUMEN_ALTO' THEN ROUND((e.cantidad_total - e.cantidad_total_esperada) / NULLIF(e.cantidad_total_esperada, 0), 4) WHEN 'APROBACION_BAJA' THEN ROUND(e.tasa_aprobacion_calc - e.tasa_aprobacion_esperada, 4) WHEN 'RECHAZO_ALTO' THEN ROUND(e.tasa_rechazo_calc - e.tasa_rechazo_esperada, 4) WHEN 'ERROR_ALTO' THEN ROUND(e.tasa_error_calc - e.tasa_error_esperada, 4) WHEN 'IMPORTE_FUERA_DE_BANDA' THEN ROUND((e.importe_total - e.importe_total_esperado) / NULLIF(e.importe_total_esperado, 0), 4) WHEN 'LATENCIA_ALTA' THEN ROUND((e.tiempo_respuesta_promedio - e.tiempo_respuesta_esperado) / NULLIF(e.tiempo_respuesta_esperado, 0), 4) WHEN 'DATO_INCONSISTENTE' THEN ROUND(e.tasa_aprobacion_informada - e.tasa_aprobacion_calc, 4) END AS desvio, e.metrica_id, e.prediccion_id FROM e WHERE e.dispara = 1;

--- 14 ---
DROP VIEW IF EXISTS v_alertas_faltantes;

--- 15 ---
CREATE VIEW v_alertas_faltantes AS WITH prm AS ( SELECT volumen_minimo, gracia_minutos, horizonte_horas, julianday(COALESCE(ahora, 'now')) AS ahora_jd FROM alerta_parametros WHERE id = 1 ) SELECT 'VENTANA_FALTANTE' AS tipo_alerta, CASE WHEN p.cantidad_total_esperada >= prm.volumen_minimo THEN 'CRITICA' ELSE 'MEDIA' END AS severidad, p.ambiente, p.cliente_id, p.servicio_id, p.producto_id, p.canal_id, p.moneda_id, p.ventana_inicio, p.duracion_ventana_minutos, 'cantidad_total' AS metrica, NULL AS valor_observado, p.cantidad_total_esperada AS valor_esperado, p.cantidad_total_minima AS umbral, -1.0 AS desvio, NULL AS metrica_id, p.prediccion_id FROM v_prediccion_vigente p CROSS JOIN prm WHERE EXISTS (SELECT 1 FROM alerta_reglas WHERE regla = 'VENTANA_FALTANTE' AND activa = 1) AND julianday(p.ventana_fin) + prm.gracia_minutos / 1440.0 <= prm.ahora_jd AND julianday(p.ventana_fin) > prm.ahora_jd - prm.horizonte_horas / 24.0 AND p.cantidad_total_esperada > 0 AND NOT EXISTS ( SELECT 1 FROM core_metricas_observadas o WHERE strftime('%Y-%m-%dT%H:%M:%S', o.ventana_inicio) = p.ventana_inicio_norm AND o.duracion_ventana_minutos = p.duracion_ventana_minutos AND o.ambiente = p.ambiente AND o.cliente_id = p.cliente_id AND o.servicio_id = p.servicio_id AND o.producto_id = p.producto_id AND o.canal_id = p.canal_id AND o.moneda_id = p.moneda_id );

--- 16 ---
CREATE VIEW v_alertas AS SELECT * FROM v_alertas_observadas UNION ALL SELECT * FROM v_alertas_faltantes;

--- 17 ---
CREATE INDEX IF NOT EXISTS ix_observadas_serie_ventana ON core_metricas_observadas ( cliente_id, servicio_id, canal_id, producto_id, moneda_id, ambiente, duracion_ventana_minutos, strftime('%Y-%m-%dT%H:%M:%S', ventana_inicio) );

===== B. HISTÓRICO (tabla si no existe, índices, v_incidentes y v_prealertas) - de alertas_d1_historico.sql =====

--- 18 ---
CREATE TABLE IF NOT EXISTS alertas_historico ( alerta_id TEXT NOT NULL DEFAULT (lower(hex(randomblob(16)))), tipo_alerta TEXT NOT NULL, ambiente TEXT NOT NULL, cliente_id TEXT NOT NULL, servicio_id TEXT NOT NULL, producto_id TEXT NOT NULL, canal_id TEXT NOT NULL, moneda_id TEXT NOT NULL, ventana_inicio TEXT NOT NULL, ventana_fin TEXT NOT NULL, duracion_ventana_minutos INTEGER NOT NULL, metrica TEXT NOT NULL, severidad TEXT NOT NULL, severidad_maxima TEXT NOT NULL, valor_observado REAL, valor_esperado REAL, umbral REAL, desvio REAL, metrica_id TEXT, prediccion_id TEXT, estado TEXT NOT NULL DEFAULT 'ABIERTA', primera_deteccion TEXT NOT NULL, ultima_deteccion TEXT NOT NULL, cantidad_detecciones INTEGER NOT NULL DEFAULT 1, cantidad_reaperturas INTEGER NOT NULL DEFAULT 0, fecha_reconocimiento TEXT, reconocida_por TEXT, fecha_resolucion TEXT, motivo_resolucion TEXT, resuelta_por TEXT, comentario TEXT, CONSTRAINT pk_alertas_historico PRIMARY KEY (alerta_id), CONSTRAINT uq_alertas_historico_clave UNIQUE ( tipo_alerta, ventana_inicio, duracion_ventana_minutos, ambiente, cliente_id, servicio_id, producto_id, canal_id, moneda_id ), CONSTRAINT ck_alertas_historico_severidad CHECK ( severidad IN ('CRITICA', 'ALTA', 'MEDIA', 'BAJA') AND severidad_maxima IN ('CRITICA', 'ALTA', 'MEDIA', 'BAJA') ), CONSTRAINT ck_alertas_historico_estado CHECK (estado IN ('ABIERTA', 'RECONOCIDA', 'RESUELTA')), CONSTRAINT ck_alertas_historico_motivo CHECK ( motivo_resolucion IS NULL OR motivo_resolucion IN ('NORMALIZADA', 'CONDICION_DESAPARECIDA', 'MANUAL') ), CONSTRAINT ck_alertas_historico_resolucion CHECK ( (estado = 'RESUELTA' AND fecha_resolucion IS NOT NULL AND motivo_resolucion IS NOT NULL) OR (estado <> 'RESUELTA' AND fecha_resolucion IS NULL AND motivo_resolucion IS NULL) ), CONSTRAINT ck_alertas_historico_ambiente CHECK (ambiente IN ('DESA', 'TEST', 'QA', 'PROD')) );

--- 19 ---
CREATE INDEX IF NOT EXISTS ix_alertas_historico_estado ON alertas_historico (estado, severidad);

--- 20 ---
CREATE INDEX IF NOT EXISTS ix_alertas_historico_ventana ON alertas_historico (ventana_inicio);

--- 21 ---
CREATE INDEX IF NOT EXISTS ix_alertas_historico_combinacion ON alertas_historico ( ambiente, cliente_id, servicio_id, producto_id, canal_id, moneda_id, duracion_ventana_minutos, tipo_alerta, ventana_inicio );

--- 22 ---
DROP VIEW IF EXISTS v_incidentes;

--- 23 ---
CREATE VIEW v_incidentes AS SELECT h.tipo_alerta, h.ambiente, h.cliente_id, h.servicio_id, h.producto_id, h.canal_id, h.moneda_id, h.duracion_ventana_minutos, CASE WHEN h.fecha_resolucion IS NULL THEN 'ABIERTO' ELSE 'RESUELTO' END AS estado_incidente, CASE MIN(CASE h.severidad_maxima WHEN 'CRITICA' THEN 1 WHEN 'ALTA' THEN 2 WHEN 'MEDIA' THEN 3 ELSE 4 END) WHEN 1 THEN 'CRITICA' WHEN 2 THEN 'ALTA' WHEN 3 THEN 'MEDIA' ELSE 'BAJA' END AS severidad_maxima, MIN(h.ventana_inicio) AS inicio, MAX(h.ventana_fin) AS ultima_ventana_afectada, h.fecha_resolucion, MAX(h.motivo_resolucion) AS motivo_resolucion, COUNT(*) AS ventanas_afectadas, MIN(h.primera_deteccion) AS primera_deteccion, ROUND( (julianday(COALESCE(h.fecha_resolucion, p.ahora)) - julianday(MIN(h.ventana_inicio))) * 1440 , 1) AS duracion_minutos FROM alertas_historico h CROSS JOIN ( SELECT strftime('%Y-%m-%dT%H:%M:%SZ', COALESCE(ahora, 'now')) AS ahora FROM alerta_parametros WHERE id = 1 ) p GROUP BY h.tipo_alerta, h.ambiente, h.cliente_id, h.servicio_id, h.producto_id, h.canal_id, h.moneda_id, h.duracion_ventana_minutos, h.fecha_resolucion;

--- 24 ---
DROP VIEW IF EXISTS v_prealertas;

--- 25 ---
CREATE VIEW v_prealertas AS WITH prm AS ( SELECT volumen_minimo, julianday(COALESCE(ahora, 'now')) AS ahora_jd FROM alerta_parametros WHERE id = 1 ), p AS ( SELECT * FROM ( SELECT e.*, strftime('%Y-%m-%dT%H:%M:%S', e.ventana_inicio) AS ventana_inicio_norm, ROW_NUMBER() OVER ( PARTITION BY strftime('%Y-%m-%dT%H:%M:%S', e.ventana_inicio), e.duracion_ventana_minutos, e.ambiente, e.cliente_id, e.servicio_id, e.producto_id, e.canal_id, e.moneda_id ORDER BY julianday(e.fecha_generacion) DESC, e.modelo_version DESC ) AS rn FROM core_metricas_esperadas e WHERE e.estado_prediccion = 'ACTIVA' AND e.ventana_inicio >= (SELECT strftime('%Y-%m-%dT%H:%M:%S', COALESCE(ahora, 'now'), '-49 hours') FROM alerta_parametros WHERE id = 1) ) WHERE rn = 1 ), c AS ( SELECT o.ambiente, o.cliente_id, o.servicio_id, o.producto_id, o.canal_id, o.moneda_id, o.duracion_ventana_minutos, strftime('%Y-%m-%dT%H:%M:%SZ', o.ventana_inicio) AS ventana_inicio, strftime('%Y-%m-%dT%H:%M:%SZ', o.ventana_fin) AS ventana_fin, o.cantidad_total, 1.0 * o.cantidad_exitos / NULLIF(o.cantidad_total, 0) AS aprob, 1.0 * o.cantidad_rechazos / NULLIF(o.cantidad_total, 0) AS rech, 1.0 * o.cantidad_errores / NULLIF(o.cantidad_total, 0) AS err, o.tiempo_respuesta_promedio AS lat, p.cantidad_total_esperada, p.cantidad_total_minima, p.tasa_aprobacion_esperada, p.tasa_aprobacion_minima, p.tasa_rechazo_esperada, p.tasa_rechazo_maxima, p.tasa_error_esperada, p.tasa_error_maxima, p.tiempo_respuesta_esperado, p.tiempo_respuesta_maximo, (p.cantidad_total_esperada - o.cantidad_total) / NULLIF(p.cantidad_total_esperada - p.cantidad_total_minima, 0) AS m_vol, CASE WHEN o.cantidad_total >= prm.volumen_minimo THEN (p.tasa_aprobacion_esperada - 1.0 * o.cantidad_exitos / o.cantidad_total) / NULLIF(p.tasa_aprobacion_esperada - p.tasa_aprobacion_minima, 0) END AS m_aprob, CASE WHEN o.cantidad_total >= prm.volumen_minimo THEN (1.0 * o.cantidad_rechazos / o.cantidad_total - p.tasa_rechazo_esperada) / NULLIF(p.tasa_rechazo_maxima - p.tasa_rechazo_esperada, 0) END AS m_rech, CASE WHEN o.cantidad_total >= prm.volumen_minimo THEN (1.0 * o.cantidad_errores / o.cantidad_total - p.tasa_error_esperada) / NULLIF(p.tasa_error_maxima - p.tasa_error_esperada, 0) END AS m_err, CASE WHEN o.cantidad_total >= prm.volumen_minimo THEN (o.tiempo_respuesta_promedio - p.tiempo_respuesta_esperado) / NULLIF(p.tiempo_respuesta_maximo - p.tiempo_respuesta_esperado, 0) END AS m_lat FROM core_metricas_observadas o JOIN p ON p.ventana_inicio_norm = strftime('%Y-%m-%dT%H:%M:%S', o.ventana_inicio) AND p.duracion_ventana_minutos = o.duracion_ventana_minutos AND p.ambiente = o.ambiente AND p.cliente_id = o.cliente_id AND p.servicio_id = o.servicio_id AND p.producto_id = o.producto_id AND p.canal_id = o.canal_id AND p.moneda_id = o.moneda_id CROSS JOIN prm WHERE o.ventana_inicio >= (SELECT strftime('%Y-%m-%dT%H:%M:%S', COALESCE(ahora, 'now'), '-49 hours') FROM alerta_parametros WHERE id = 1) AND julianday(o.ventana_fin) <= prm.ahora_jd AND o.cantidad_total > 0 ), t AS ( SELECT c.*, LAG(c.ventana_inicio, 2) OVER s AS ventana_inicio_2, MAX(c.ventana_inicio) OVER (PARTITION BY c.ambiente, c.cliente_id, c.servicio_id, c.producto_id, c.canal_id, c.moneda_id, c.duracion_ventana_minutos) AS ultima_ventana_serie, LAG(c.m_vol, 1) OVER s AS m_vol_1, LAG(c.m_vol, 2) OVER s AS m_vol_2, LAG(c.m_aprob, 1) OVER s AS m_aprob_1, LAG(c.m_aprob, 2) OVER s AS m_aprob_2, LAG(c.m_rech, 1) OVER s AS m_rech_1, LAG(c.m_rech, 2) OVER s AS m_rech_2, LAG(c.m_err, 1) OVER s AS m_err_1, LAG(c.m_err, 2) OVER s AS m_err_2, LAG(c.m_lat, 1) OVER s AS m_lat_1, LAG(c.m_lat, 2) OVER s AS m_lat_2 FROM c WINDOW s AS (PARTITION BY c.ambiente, c.cliente_id, c.servicio_id, c.producto_id, c.canal_id, c.moneda_id, c.duracion_ventana_minutos ORDER BY c.ventana_inicio) ), x AS ( SELECT t.*, k.column1 AS tipo, k.column2 AS tipo_previsto, k.column3 AS metrica, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err ELSE t.m_lat END AS m, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol_1 WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob_1 WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech_1 WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err_1 ELSE t.m_lat_1 END AS m1, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.m_vol_2 WHEN 'RIESGO_APROBACION_BAJA' THEN t.m_aprob_2 WHEN 'RIESGO_RECHAZO_ALTO' THEN t.m_rech_2 WHEN 'RIESGO_ERROR_ALTO' THEN t.m_err_2 ELSE t.m_lat_2 END AS m2, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total WHEN 'RIESGO_APROBACION_BAJA' THEN ROUND(t.aprob, 4) WHEN 'RIESGO_RECHAZO_ALTO' THEN ROUND(t.rech, 4) WHEN 'RIESGO_ERROR_ALTO' THEN ROUND(t.err, 4) ELSE t.lat END AS valor_observado, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total_esperada WHEN 'RIESGO_APROBACION_BAJA' THEN t.tasa_aprobacion_esperada WHEN 'RIESGO_RECHAZO_ALTO' THEN t.tasa_rechazo_esperada WHEN 'RIESGO_ERROR_ALTO' THEN t.tasa_error_esperada ELSE t.tiempo_respuesta_esperado END AS valor_esperado, CASE k.column1 WHEN 'RIESGO_VOLUMEN_BAJO' THEN t.cantidad_total_minima WHEN 'RIESGO_APROBACION_BAJA' THEN t.tasa_aprobacion_minima WHEN 'RIESGO_RECHAZO_ALTO' THEN t.tasa_rechazo_maxima WHEN 'RIESGO_ERROR_ALTO' THEN t.tasa_error_maxima ELSE t.tiempo_respuesta_maximo END AS umbral FROM t CROSS JOIN (VALUES ('RIESGO_VOLUMEN_BAJO', 'VOLUMEN_BAJO', 'cantidad_total'), ('RIESGO_APROBACION_BAJA', 'APROBACION_BAJA', 'tasa_aprobacion'), ('RIESGO_RECHAZO_ALTO', 'RECHAZO_ALTO', 'tasa_rechazo'), ('RIESGO_ERROR_ALTO', 'ERROR_ALTO', 'tasa_error'), ('RIESGO_LATENCIA_ALTA', 'LATENCIA_ALTA', 'tiempo_respuesta_promedio') ) AS k ), senal AS ( SELECT x.*, CASE WHEN x.m + (x.m - x.m2) / 2 >= 1 THEN 'ALTA' ELSE 'MEDIA' END AS severidad, CAST(strftime('%s', x.ventana_inicio) AS INTEGER) / (60 * x.duracion_ventana_minutos) - ROW_NUMBER() OVER (PARTITION BY x.ambiente, x.cliente_id, x.servicio_id, x.producto_id, x.canal_id, x.moneda_id, x.duracion_ventana_minutos, x.tipo ORDER BY x.ventana_inicio) AS grupo FROM x WHERE ROUND((julianday(x.ventana_inicio) - julianday(x.ventana_inicio_2)) * 1440) = 2 * x.duracion_ventana_minutos AND COALESCE(x.m_vol, 0) < 1 AND COALESCE(x.m_aprob, 0) < 1 AND COALESCE(x.m_rech, 0) < 1 AND COALESCE(x.m_err, 0) < 1 AND COALESCE(x.m_lat, 0) < 1 AND x.m >= 0.5 AND x.m - x.m1 >= 0.1 AND x.m1 - x.m2 >= 0.1 AND x.m + (x.m - x.m2) >= 1 ), ep AS ( SELECT s.*, MIN(s.ventana_inicio) OVER g AS inicio, COUNT(*) OVER g AS ventanas, MAX(s.m) OVER g AS margen_max, ROW_NUMBER() OVER (PARTITION BY s.ambiente, BY s.ambiente, s.cliente_id, s.servicio_id, s.producto_id, s.canal_id, s.moneda_id, s.duracion_ventana_minutos, s.tipo, s.grupo) ), cruce AS ( SELECT ep.*, (SELECT MIN(a.ventana_fin) FROM alertas_historico a WHERE a.ambiente = ep.ambiente AND a.cliente_id = ep.cliente_id AND a.servicio_id = ep.servicio_id AND a.producto_id = ep.producto_id AND a.canal_id = ep.canal_id AND a.moneda_id = julianday(ep.ventana_fin) + 60 / 1440.0 ) AS alerta_real_ventana_fin FROM ep WHERE ep.desde_el_final = 1 ) SELECT cruce.tipo AS tipo_prealerta, cruce.tipo_previsto, cruce.ambiente, cruce.cliente_id, cruce.servicio_id, cruce.producto_id, cruce.canal_id, cruce.moneda_id, cruce.duracion_ventana_minutos, cruce.metrica, cruce.inicio, cruce.ventana_inicio AS ultima_ventana, cruce.ventanas, cruce.severidad, cruce.valor_observado, cruce.valor_esperado, cruce.umbral, ROUND(cruce.m, 2) AS margen_consumido, ROUND(cruce.margen_max, 2) AS margen_consumido_max, CASE cruce.severidad WHEN 'ALTA' THEN 1 ELSE 2 END * cruce.duracion_ventana_minutos AS minutos_estimados, CASE WHEN cruce.ventana_inicio = cruce.ultima_ventana_serie AND cruce.alerta_real_ventana_fin IS NULL AND julianday(cruce.ventana_fin) >= prm.ahora_jd - 3 * cruce.duracion_ventana_minutos / 1440.0 THEN 1 ELSE 0 END AS activa, CASE WHEN cruce.alerta_real_ventana_fin IS NOT NULL THEN 'CONFIRMADA' WHEN prm.ahora_jd < julianday(cruce.ventana_fin) + 60 / 1440.0 THEN 'EN_OBSERVACION' ELSE 'NO_SE_CUMPLIO' END AS resultado, cruce.alerta_real_ventana_fin, ROUND((julianday(cruce.alerta_real_ventana_fin) - julianday(cruce.inicio) - cruce.duracion_ventana_minutos / 1440.0) * 1440) AS minutos_anticipacion FROM cruce CROSS JOIN prm;
